# ---------------------------------------------------------------
# LiveRecorder 运行依赖
#
# 安装：
#     pip install -r requirements.txt
#
# 说明：录制核心依赖外部的 ffmpeg 可执行文件（不通过 pip 安装），
#      安装方式见 README.md「安装步骤 → 安装 ffmpeg」。
# ---------------------------------------------------------------

# 图形界面
PyQt6>=6.5.0

# HTTP 客户端（带连接池、Cookie 管理、HTTP/2 支持）
httpx>=0.25.0

# 备用 HTTP 客户端（部分场景下更易做流式处理）
requests>=2.31.0

# Brotli 解码支持（重要）
# 两个平台的网页响应常以 br 压缩返回。若声明了 Accept-Encoding: br 却没有该解码器，
# httpx 会把压缩字节当作文本返回（不报错），导致「接口 200 但解析不到任何字段」。
brotli>=1.1.0

# 可选：处理带动态签名参数的平台接口（抖音等）
# 启用需额外执行：playwright install chromium
playwright>=1.40.0
